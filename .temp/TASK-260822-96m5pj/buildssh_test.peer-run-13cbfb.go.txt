// buildSSHConfig carries all three authentication shapes: identity only,
// agent only, and an agent pinned to one identity.
const buildSSHConfig = `{
	"schema_version": 1, "skills_root": "/tmp/skills", "projects": {},
	"build_ssh": {
		"git.example.com": {"agent": "/run/agent.sock"},
		"git.example.com/relux-works": {"identity": "~/.ssh/org", "known_hosts": "~/.ssh/known_hosts_org"},
		"git.example.com/relux-works/portals": {"agent": "/run/portals.sock", "identity": "/keys/portals"}
	}
}`

func loadBuildSSH(t *testing.T) *Config {
	t.Helper()
	cfg, err := Load(writeConfig(t, t.TempDir(), "config.json", buildSSHConfig), nil)
	if err != nil {
		t.Fatal(err)
	}
	return cfg
}

func TestParseBuildSSH(t *testing.T) {
	cfg := loadBuildSSH(t)
	want := map[string]BuildSSHCredential{
		"git.example.com": {
			Scope: "git.example.com", Agent: "/run/agent.sock",
		},
		"git.example.com/relux-works": {
			Scope: "git.example.com/relux-works", Identity: "~/.ssh/org", KnownHosts: "~/.ssh/known_hosts_org",
		},
		"git.example.com/relux-works/portals": {
			Scope: "git.example.com/relux-works/portals", Agent: "/run/portals.sock", Identity: "/keys/portals",
		},
	}
	if !reflect.DeepEqual(cfg.BuildSSH, want) {
		t.Fatalf("build_ssh = %+v, want %+v", cfg.BuildSSH, want)
	}
	wantScopes := []string{"git.example.com", "git.example.com/relux-works", "git.example.com/relux-works/portals"}
	if got := cfg.BuildSSHScopes(); !reflect.DeepEqual(got, wantScopes) {
		t.Fatalf("scopes = %v, want %v", got, wantScopes)
	}
}

func TestParseBuildSSHAbsent(t *testing.T) {
	for name, text := range map[string]string{
		"absent": minimal,
		"null":   `{"schema_version": 1, "skills_root": "x", "projects": {}, "build_ssh": null}`,
		"empty":  `{"schema_version": 1, "skills_root": "x", "projects": {}, "build_ssh": {}}`,
	} {
		t.Run(name, func(t *testing.T) {
			cfg, err := Load(writeConfig(t, t.TempDir(), "config.json", text), nil)
			if err != nil {
				t.Fatal(err)
			}
			if len(cfg.BuildSSH) != 0 {
				t.Fatalf("build_ssh = %+v, want empty", cfg.BuildSSH)
			}
			if len(cfg.BuildSSHScopes()) != 0 {
				t.Fatalf("scopes = %v, want none", cfg.BuildSSHScopes())
			}
			if _, ok := cfg.BuildSSHFor("git.example.com/relux-works/portals"); ok {
				t.Fatal("an unset build_ssh must select no credential")
			}
		})
	}
}

// TestBuildSSHSerializationRoundtrip writes the parsed credentials back out
// through BuildSSHObject and requires the reparsed config to be identical.
func TestBuildSSHSerializationRoundtrip(t *testing.T) {
	dir := t.TempDir()
	first := loadBuildSSH(t)
	payload, err := json.Marshal(map[string]any{
		"schema_version": 1, "skills_root": "/tmp/skills", "projects": map[string]any{},
		"build_ssh": BuildSSHObject(first.BuildSSH),
	})
	if err != nil {
		t.Fatal(err)
	}
	path := filepath.Join(dir, "roundtrip.json")
	if err := os.WriteFile(path, payload, 0o600); err != nil {
		t.Fatal(err)
	}
	second, err := Load(path, nil)
	if err != nil {
		t.Fatalf("serialized build_ssh did not parse: %v", err)
	}
	if !reflect.DeepEqual(second.BuildSSH, first.BuildSSH) {
		t.Fatalf("roundtrip = %+v, want %+v", second.BuildSSH, first.BuildSSH)
	}
}

func TestBuildSSHObjectShape(t *testing.T) {
	object := BuildSSHObject(map[string]BuildSSHCredential{
		"a.example.com": {Scope: "a.example.com", Agent: "/run/agent.sock"},
		"b.example.com": {Scope: "b.example.com", Identity: "/keys/id", KnownHosts: "/etc/known_hosts"},
	})
	payload, err := json.Marshal(object)
	if err != nil {
		t.Fatal(err)
	}
	want := `{"a.example.com":{"agent":"/run/agent.sock"},` +
		`"b.example.com":{"identity":"/keys/id","known_hosts":"/etc/known_hosts"}}`
	if string(payload) != want {
		t.Fatalf("serialized = %s, want %s", payload, want)
	}
	if got := BuildSSHObject(nil); got != nil {
		t.Fatalf("BuildSSHObject(nil) = %v, want nil", got)
	}
}

func TestValidBuildSSHScope(t *testing.T) {
	cases := []struct {
		scope string
		want  bool
	}{
		{"git.example.com", true},
		{"git.example.com/relux-works", true},
		{"git.example.com/relux-works/portals", true},
		{"git.example.com/a.b/c_d/e-f", true},
		// The canonical identity path keeps its case (Spec §6.3), so an
		// upper-case path segment stays addressable.
		{"git.example.com/RELUX", true},
		{"localhost", true},
		{"127.0.0.1/repo", true},
		{strings.Repeat("a", 253), true},
		{"", false},
		{"Git.Example.com", false},
		{"GIT.EXAMPLE.COM/repo", false},
		{"git.example.com//portals", false},
		{"git.example.com/", false},
		{"/git.example.com", false},
		{"git.example.com/.", false},
		{"git.example.com/..", false},
		{"git.example.com/a b", false},
		{"git.example.com/a\\b", false},
		{"git.example.com/a:b", false},
		{"git.example.com/ünicode", false},
		{"git..example.com", false},
		{"git.example.com.", false},
		{"-git.example.com", false},
		{"git.example.com:22/repo", false},
		{"ssh://git.example.com/repo", false},
		{"git@example.com:repo", false},
		{strings.Repeat("a", 254), false},
	}
	for _, tc := range cases {
		if got := ValidBuildSSHScope(tc.scope); got != tc.want {
			t.Errorf("ValidBuildSSHScope(%q) = %v, want %v", tc.scope, got, tc.want)
		}
	}
}

func TestParseBuildSSHRejections(t *testing.T) {
	const prefix = `{"schema_version": 1, "skills_root": "x", "projects": {}, "build_ssh": `
	cases := []struct {
		name string
		body string
		want string
	}{
		{"not an object", `[]`, "build_ssh: must be an object"},
		{"entry not an object", `{"git.example.com": "/run/agent.sock"}`, "build_ssh.git.example.com: must be an object"},
		{"uppercase host", `{"Git.Example.com": {"agent": "/a"}}`, `scope "Git.Example.com"`},
		{"empty segment", `{"git.example.com//portals": {"agent": "/a"}}`, `scope "git.example.com//portals"`},
		{"trailing slash", `{"git.example.com/portals/": {"agent": "/a"}}`, `scope "git.example.com/portals/"`},
		{"dot segment", `{"git.example.com/../portals": {"agent": "/a"}}`, `scope "git.example.com/../portals"`},
		{"scheme in scope", `{"ssh://git.example.com": {"agent": "/a"}}`, "scope"},
		{"empty scope", `{"": {"agent": "/a"}}`, `scope ""`},
		{"unknown entry field", `{"git.example.com": {"agent": "/a", "port": 22, "user": "git"}}`, "unsupported field(s): port, user"},
		{"no selection", `{"git.example.com": {"known_hosts": "/k"}}`, "requires 'agent', 'identity', or both"},
		{"empty entry", `{"git.example.com": {}}`, "requires 'agent', 'identity', or both"},
		{"agent null", `{"git.example.com": {"agent": null}}`, "build_ssh.git.example.com.agent"},
		{"agent true", `{"git.example.com": {"agent": true}}`, "build_ssh.git.example.com.agent"},
		{"agent number", `{"git.example.com": {"agent": 22}}`, "build_ssh.git.example.com.agent"},
		{"agent empty", `{"git.example.com": {"agent": "", "identity": "/keys/id"}}`, "build_ssh.git.example.com.agent"},
		{"agent relative", `{"git.example.com": {"agent": "agent.sock"}}`, "build_ssh.git.example.com.agent"},
		{"identity null", `{"git.example.com": {"identity": null}}`, "build_ssh.git.example.com.identity"},
		{"identity empty", `{"git.example.com": {"identity": ""}}`, "build_ssh.git.example.com.identity"},
		{"identity relative", `{"git.example.com": {"identity": ".ssh/id"}}`, "build_ssh.git.example.com.identity"},
		{"identity bare tilde", `{"git.example.com": {"identity": "~other/id"}}`, "build_ssh.git.example.com.identity"},
		{"identity control byte", `{"git.example.com": {"identity": "/keys/\u0000id"}}`, "build_ssh.git.example.com.identity"},
		{"known_hosts empty", `{"git.example.com": {"agent": "/a", "known_hosts": ""}}`, "build_ssh.git.example.com.known_hosts"},
		{"known_hosts relative", `{"git.example.com": {"agent": "/a", "known_hosts": "known_hosts"}}`, "build_ssh.git.example.com.known_hosts"},
		{"known_hosts number", `{"git.example.com": {"agent": "/a", "known_hosts": 1}}`, "build_ssh.git.example.com.known_hosts"},
	}
	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			_, err := Load(writeConfig(t, t.TempDir(), "config.json", prefix+tc.body+"}"), nil)
			if err == nil || !strings.Contains(err.Error(), tc.want) {
				t.Fatalf("err = %v, want mention of %q", err, tc.want)
			}
		})
	}
}

// TestParseBuildSSHReportsAStableScope pins the reported scope so a config
// with several bad scopes always fails the same way instead of following Go's
// randomized map order.
func TestParseBuildSSHReportsAStableScope(t *testing.T) {
	text := `{"schema_version": 1, "skills_root": "x", "projects": {}, "build_ssh": {
		"Zeta.example.com": {"agent": "/a"},
		"Alpha.example.com": {"agent": "/a"},
		"Mid.example.com": {"agent": "/a"}
	}}`
	path := writeConfig(t, t.TempDir(), "config.json", text)
	for range 16 {
		_, err := Load(path, nil)
		if err == nil || !strings.Contains(err.Error(), `scope "Alpha.example.com"`) {
			t.Fatalf("err = %v, want the lexicographically first bad scope", err)
		}
	}
}

func TestBuildSSHLongestPrefixMatch(t *testing.T) {
	cfg := loadBuildSSH(t)
	cases := []struct {
		identity string
		want     string // matched scope, "" for no match
	}{
		{"git.example.com/relux-works/portals", "git.example.com/relux-works/portals"},
		{"git.example.com/relux-works/portals/sub", "git.example.com/relux-works/portals"},
		// Boundary: a scope matches whole segments only, so the longer
		// repository name falls back to the enclosing organization scope
		// instead of borrowing the portals credential.
		{"git.example.com/relux-works/portals-evil", "git.example.com/relux-works"},
		{"git.example.com/relux-works/portals.evil", "git.example.com/relux-works"},
		{"git.example.com/relux-works/portalsevil", "git.example.com/relux-works"},
		{"git.example.com/relux-works", "git.example.com/relux-works"},
		{"git.example.com/relux-works-evil/portals", "git.example.com"},
		{"git.example.com/other/repo", "git.example.com"},
		{"git.example.community/relux-works/portals", ""},
		{"other.example.com/relux-works/portals", ""},
		{"", ""},                // a local source carries no identity
		{"git.example.com", ""}, // a host alone is not a repository identity
		{"GIT.EXAMPLE.COM/relux-works/portals", ""},
		{"ssh://git.example.com/relux-works/portals", ""},
		{"git.example.com/relux-works/../portals", ""},
	}
	for _, tc := range cases {
		got, ok := cfg.BuildSSHFor(tc.identity)
		if tc.want == "" {
			if ok {
				t.Errorf("BuildSSHFor(%q) = %+v, want no match", tc.identity, got)
			}
			continue
		}
		if !ok || got.Scope != tc.want {
			t.Errorf("BuildSSHFor(%q) = %+v, %v; want scope %q", tc.identity, got, ok, tc.want)
		}
	}
}

func TestBuildSSHForCarriesTheWholeCredential(t *testing.T) {
	cfg := loadBuildSSH(t)
	credential, ok := cfg.BuildSSHFor("git.example.com/relux-works/portals/deep")
	want := BuildSSHCredential{
		Scope: "git.example.com/relux-works/portals",
		Agent: "/run/portals.sock", Identity: "/keys/portals",
	}
	if !ok || credential != want {
		t.Fatalf("credential = %+v, %v; want %+v", credential, ok, want)
	}
	if credential.Empty() {
		t.Fatal("a pinned-agent credential must not read as empty")
	}
	if !(BuildSSHCredential{Scope: "git.example.com", KnownHosts: "/k"}).Empty() {
		t.Fatal("known_hosts alone selects no credential and must read as empty")
	}
}

func TestBuildSSHExpandedResolvesHome(t *testing.T) {
	home, err := os.UserHomeDir()
	if err != nil {
		t.Skipf("no home directory: %v", err)
	}
	credential := loadBuildSSH(t).BuildSSH["git.example.com/relux-works"]
	expanded := credential.Expanded()
	if want := filepath.Join(home, ".ssh", "org"); expanded.Identity != want {
		t.Errorf("Identity = %q, want %q", expanded.Identity, want)
	}
	if want := filepath.Join(home, ".ssh", "known_hosts_org"); expanded.KnownHosts != want {
		t.Errorf("KnownHosts = %q, want %q", expanded.KnownHosts, want)
	}
	// The stored spelling stays portable so serialization stays stable.
	if credential.Identity != "~/.ssh/org" {
		t.Errorf("Identity = %q, want the operator spelling preserved", credential.Identity)
	}
	absolute := BuildSSHCredential{
		Scope: "git.example.com", Agent: "/run/agent.sock",
		Identity: "/keys/id", KnownHosts: "/etc/known_hosts",
	}
	if got := absolute.Expanded(); got != absolute {
		t.Errorf("Expanded(%+v) = %+v, want unchanged", absolute, got)
	}
}

func TestValidateBuildSSH(t *testing.T) {
	cases := []struct {
		name       string
		credential BuildSSHCredential
		wantErr    string
	}{
		{"agent only", BuildSSHCredential{Scope: "git.example.com", Agent: "/run/a.sock"}, ""},
		{"identity only", BuildSSHCredential{Scope: "git.example.com/org", Identity: "~/.ssh/id"}, ""},
		{"pinned agent", BuildSSHCredential{Scope: "git.example.com/org/repo", Agent: "/run/a.sock", Identity: "/keys/id"}, ""},
		{"windows identity", BuildSSHCredential{Scope: "git.example.com", Identity: `C:\keys\id`}, ""},
		{"unc identity", BuildSSHCredential{Scope: "git.example.com", Identity: `\\host\share\id`}, ""},
		{"bad scope", BuildSSHCredential{Scope: "Git.Example.com", Agent: "/run/a.sock"}, `scope "Git.Example.com"`},
		{"no scope", BuildSSHCredential{Agent: "/run/a.sock"}, `scope ""`},
		{"empty credential", BuildSSHCredential{Scope: "git.example.com"}, "requires 'agent', 'identity', or both"},
		{"known_hosts alone", BuildSSHCredential{Scope: "git.example.com", KnownHosts: "/etc/known_hosts"}, "requires 'agent', 'identity', or both"},
		{"relative agent", BuildSSHCredential{Scope: "git.example.com", Agent: "agent.sock"}, "build_ssh.git.example.com.agent"},
		{"relative identity", BuildSSHCredential{Scope: "git.example.com", Identity: "keys/id"}, "build_ssh.git.example.com.identity"},
		{"relative known_hosts", BuildSSHCredential{Scope: "git.example.com", Agent: "/run/a.sock", KnownHosts: "known_hosts"}, "build_ssh.git.example.com.known_hosts"},
		{"bare tilde", BuildSSHCredential{Scope: "git.example.com", Identity: "~"}, "build_ssh.git.example.com.identity"},
		{"control character", BuildSSHCredential{Scope: "git.example.com", Identity: "/keys/\tid"}, "build_ssh.git.example.com.identity"},
		{"overlong identity", BuildSSHCredential{Scope: "git.example.com", Identity: "/" + strings.Repeat("k", maxCredentialPathLength)}, "build_ssh.git.example.com.identity"},
	}
	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			err := ValidateBuildSSH(tc.credential)
			if tc.wantErr == "" {
				if err != nil {
					t.Fatalf("err = %v, want nil", err)
				}
				return
			}
			if err == nil || !strings.Contains(err.Error(), tc.wantErr) {
				t.Fatalf("err = %v, want mention of %q", err, tc.wantErr)
			}
		})
	}
}

// TestValidateBuildSSHAcceptsEveryParsedCredential keeps the standalone
// validator used by config writers aligned with the parser.
func TestValidateBuildSSHAcceptsEveryParsedCredential(t *testing.T) {
	for scope, credential := range loadBuildSSH(t).BuildSSH {
		if err := ValidateBuildSSH(credential); err != nil {
			t.Errorf("ValidateBuildSSH(%q, %+v) = %v", scope, credential, err)
		}
	}
}

func TestBuildSSHIsAManagerKey(t *testing.T) {
	if !managerKeys["build_ssh"] {
		t.Fatal("build_ssh must be a known top-level key")
	}
	dir := t.TempDir()
	userPath := writeConfig(t, dir, "config.json", minimal)
	systemPath := writeConfig(t, dir, "system.json",
		`{"schema_version": 1, "build_ssh": {"git.example.com": {"agent": "/run/agent.sock"}}}`)
	t.Setenv("CURATOR_SYSTEM_CONFIG", systemPath)
	cfg, err := Load(userPath, nil)
	if err != nil {
		t.Fatal(err)
	}
	if _, ok := cfg.BuildSSHFor("git.example.com/org/repo"); !ok {
		t.Fatalf("system build_ssh default was not applied: %+v", cfg.BuildSSH)
	}
}
