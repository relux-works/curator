// Review probe B: a lock resolved from an empty selection, then a manifest
// that declares a skill without refresh. The lock is stale and the declared
// skill is neither resolved nor installed: `status --check` must be nonzero.
func TestReviewDraftStatusStaleEmptyLockIsNotCurrent(t *testing.T) {
	withDraftSourcesSwitch(t)
	root := t.TempDir()
	configPath, project := setupCLIProject(t, root)
	empty := `{"schema_version":2,"sources":{"s":{"path":"."}},"skills":[]}`
	if err := os.WriteFile(filepath.Join(project, "Skillfile.json"), []byte(empty), 0o644); err != nil {
		t.Fatal(err)
	}
	writeCLISkill(t, filepath.Join(project, "skills", "review"), "review")
	code, stdout, stderr := capture(t, configPath, "project", "resolve", "app")
	t.Logf("resolve(empty) code=%d\nstdout:%s\nstderr:%s", code, stdout, stderr)
	if code != exitOK {
		t.Skipf("an empty selection does not resolve (code %d): the zero-member lock is unreachable through the CLI", code)
	}
	lock, err := sourcelock.Read(sourcelock.PathIn(project))
	if err != nil {
		t.Fatal(err)
	}
	t.Logf("lock members = %d", len(lock.Members))
	if err := os.WriteFile(filepath.Join(project, "Skillfile.json"), []byte(draftStatusPayload), 0o644); err != nil {
		t.Fatal(err)
	}
	code, stdout, stderr = capture(t, configPath, "status", "app")
	t.Logf("status code=%d\nstdout:%s\nstderr:%s", code, stdout, stderr)
	code, stdout, stderr = capture(t, configPath, "status", "--check", "app")
	t.Logf("status --check code=%d\nstdout:%s\nstderr:%s", code, stdout, stderr)
	if code == exitOK {
		t.Fatalf("status --check = 0 for a stale lock whose manifest now declares an uninstalled skill")
	}
}

// Review probe C: with members in the lock, a manifest edit that swaps the
// declared skill (review -> other) without refresh. Records what the rows
// and diagnostics say; asserts only that --check is nonzero.
func TestReviewDraftStatusStaleLockRowsAndDiagnostics(t *testing.T) {
	configPath, project := setupDraftStatusProject(t)
	runInstall(t, configPath)
	writeCLISkill(t, filepath.Join(project, "skills", "other"), "other")
	swapped := `{"schema_version":2,"sources":{"s":{"path":"."}},"skills":[{"from":"s","directory":"skills","include":["other"]}]}`
	if err := os.WriteFile(filepath.Join(project, "Skillfile.json"), []byte(swapped), 0o644); err != nil {
		t.Fatal(err)
	}
	code, stdout, stderr := capture(t, configPath, "status", "app")
	t.Logf("status code=%d\nstdout:%s\nstderr:%s", code, stdout, stderr)
	code, stdout, stderr = capture(t, configPath, "status", "--json", "app")
	t.Logf("status --json code=%d\nstdout:%s\nstderr:%s", code, stdout, stderr)
	code, _, _ = capture(t, configPath, "status", "--check", "app")
	if code == exitOK {
		t.Fatalf("status --check = 0 for a stale lock")
	}
}

// Review probe D: a lock-only change (a second member joins the selection;
// review's package is unchanged) must make review needs-install through the
// CLI, because its marker binds the previous lock generation.
func TestReviewDraftStatusLockOnlyChangeNeedsInstall(t *testing.T) {
	configPath, project := setupDraftStatusProject(t)
	runInstall(t, configPath)
	writeCLISkill(t, filepath.Join(project, "skills", "other"), "other")
	widened := `{"schema_version":2,"sources":{"s":{"path":"."}},"skills":[{"from":"s","directory":"skills","include":["review","other"]}]}`
	if err := os.WriteFile(filepath.Join(project, "Skillfile.json"), []byte(widened), 0o644); err != nil {
		t.Fatal(err)
	}
	if code, _, stderr := capture(t, configPath, "project", "refresh", "app"); code != exitOK {
		t.Fatalf("project refresh = %d\nstderr:\n%s", code, stderr)
	}
	code, stdout, stderr := capture(t, configPath, "status", "app")
	t.Logf("status code=%d\nstdout:%s\nstderr:%s", code, stdout, stderr)
	if !strings.Contains(stdout, "app: review needs-install") {
		t.Fatalf("review must be needs-install after a lock-only change:\n%s", stdout)
	}
	if code, _, _ := capture(t, configPath, "status", "--check", "app"); code == exitOK {
		t.Fatalf("status --check = 0 after a lock-only change")
	}
}
