## Overlay-only review test appended to internal/envprofile/muse_credential_policy_test.go

// Review attack: an isolated vendor credential survives native-to-file selector
// changes, and shared-mode repair refuses rather than replacing it with a link.
func TestReviewMuseIsolatedCredentialSwitch(t *testing.T) {
 fx := writeManagedFixture(t, "review-muse")
 fx.native[envregistry.Muse] = t.TempDir()
 native := filepath.Join(fx.native[envregistry.Muse], "auth.json")
 if err := os.WriteFile(native, []byte("synthetic-native\n"), 0600); err != nil { t.Fatal(err) }
 req := fx.request(envregistry.Muse)
 req.Repair = true
 req.Machine.Isolation[fx.profile] = map[string]string{envregistry.Muse:"isolated"}
 if _,err := Resolve(req); err != nil { t.Fatal(err) }
 auth := filepath.Join(ManagedHomeDir(fx.home,fx.profile,envregistry.Muse),"config","muse","auth.json")
 if err := os.WriteFile(auth,[]byte("synthetic-isolated\n"),0600); err != nil {t.Fatal(err)}
 req.Machine.CredentialStore = map[string]map[string]envregistry.CredentialStoreSelection{fx.profile:{envregistry.Muse:envregistry.CredentialStoreFile}}
 before := snapshotCredentialScope(t,fx)
 if _,err := Resolve(req); err != nil {t.Fatalf("compatible selector switch refused: %v",err)}
 if !reflect.DeepEqual(before,snapshotCredentialScope(t,fx)) {t.Fatal("compatible switch altered credential state")}
 req.Machine.Isolation[fx.profile][envregistry.Muse]="shared"
 for _,repair := range []bool{false,true} {
  req.Repair=repair
  result,err:=Resolve(req)
  if err==nil || result!=nil && len(result.Document)>0 {t.Fatalf("shared switch admitted vendor credential: repair=%v err=%v",repair,err)}
  if repair && !strings.Contains(err.Error(),envregistry.DiagCredentialConflict) {t.Fatalf("wrong repair refusal: %v",err)}
  if !reflect.DeepEqual(before,snapshotCredentialScope(t,fx)) {t.Fatal("shared switch altered credential state")}
 }
}
