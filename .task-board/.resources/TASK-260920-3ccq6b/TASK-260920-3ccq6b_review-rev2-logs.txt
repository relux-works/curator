		case strings.HasPrefix(line, pins+"<fetch>"):
			fetch++
		case strings.HasPrefix(line, pins+"<remote>"):
			remote++
		case strings.Contains(line, "<clone>") || strings.Contains(line, "<fetch>"):
			other++
			t.Logf("unpinned network invocation: %s", line)
		}
	}
	t.Logf("pinned clone=%d fetch=%d remote=%d; unpinned clone/fetch=%d", clone, fetch, remote, other)
	if clone != 1 || fetch != 1 || other != 0 {
		t.Fatalf("pins missing at the production entry: clone=%d fetch=%d unpinned=%d\n%s", clone, fetch, other, argv)
	}
}
